/**
 * Checks if a license number is in the ELFA network
 */

  const cleanLicense = licenseNumber.trim();
  const isElfa = ELFA_FCC_LICENSES.has(cleanLicense);
  console.log(`ELFA check: ${cleanLicense} = ${isElfa}`);
